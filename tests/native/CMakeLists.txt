cmake_minimum_required(VERSION 3.5)
project(SDIO_Integration_Policy_Tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(test_plan ../../native/plan_validation.cpp test_plan.cpp)
target_include_directories(test_plan PRIVATE ../../native)
add_test(NAME installation_plan_validation COMMAND test_plan)
add_executable(test_unavailable ../../native/origin_unavailable.cpp test_unavailable.cpp)
target_include_directories(test_unavailable PRIVATE ../../native)
add_test(NAME unavailable_backend_cannot_fake_capabilities COMMAND test_unavailable)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(test_plan PRIVATE -Wall -Wextra -Werror)
  target_compile_options(test_unavailable PRIVATE -Wall -Wextra -Werror)
endif()
