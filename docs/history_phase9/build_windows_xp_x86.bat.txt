@echo off
setlocal
REM Run from a Visual Studio 2013 x86 command prompt with the XP-targeting SDK.
REM QMAKE must refer to your STATIC Qt 5.6.3 build configured with -target xp.
cd /d "%~dp0.."
if not defined QMAKE set "QMAKE=qmake"
"%QMAKE%" -query QT_VERSION >nul 2>nul
if errorlevel 1 (
  echo ERROR: qmake is not available. Set QMAKE to Qt 5.6.3 qmake.exe.
  exit /b 1
)
for /f "delims=" %%v in ('"%QMAKE%" -query QT_VERSION') do set "QTVERSION=%%v"
echo Qt version: %QTVERSION%
if not "%QTVERSION%"=="5.6.3" (
  echo ERROR: The XP build requires Qt 5.6.3, not %QTVERSION%.
  exit /b 1
)
if not exist "build-xp-x86" mkdir "build-xp-x86"
cd build-xp-x86
%QMAKE% ..\SDIO_Modern_Qt.pro "CONFIG+=release" "CONFIG-=debug"
if errorlevel 1 exit /b 1
where nmake >nul 2>nul
if errorlevel 1 (
  echo ERROR: This script expects a Visual Studio x86 command prompt with nmake.
  exit /b 1
)
nmake /NOLOGO
if errorlevel 1 exit /b 1
if exist release\SDIO-Modern-Qt.exe (
  echo BUILD COMPLETED: build-xp-x86\release\SDIO-Modern-Qt.exe
  echo VERIFY: Qt was linked statically and PE imports work on Windows XP SP3.
) else if exist SDIO-Modern-Qt.exe (
  echo BUILD COMPLETED: build-xp-x86\SDIO-Modern-Qt.exe
  echo VERIFY: Qt was linked statically and PE imports work on Windows XP SP3.
) else (
  echo ERROR: Missing final EXE. Inspect the build log.
  exit /b 1
)
endlocal
